cmake_minimum_required(VERSION 3.22)
project(leaf_simulator LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Configure with emcmake cmake -S simulator -B simulator/build")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
get_filename_component(LEAF_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(LVGL_DIR "${LEAF_ROOT}/managed_components/lvgl__lvgl" CACHE PATH "Firmware LVGL 9.2.2 source")
if(NOT EXISTS "${LVGL_DIR}/lvgl.h")
  include(FetchContent)
  FetchContent_Declare(lvgl_src GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.2.2 GIT_SHALLOW TRUE)
  FetchContent_GetProperties(lvgl_src)
  if(NOT lvgl_src_POPULATED)
    FetchContent_Populate(lvgl_src)
  endif()
  set(LVGL_DIR "${lvgl_src_SOURCE_DIR}")
endif()
set(UI "${LEAF_ROOT}/main/justtype")
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL_DIR}" "${UI}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE)
# Deliberately compile the firmware sources in place, without copied UI code.
file(GLOB UI_SOURCES CONFIGURE_DEPENDS "${UI}/src/*.cpp" "${UI}/src/icons/*.cpp" "${UI}/src/spelling/*.cpp" "${UI}/src/fonts/*.c")
list(REMOVE_ITEM UI_SOURCES "${UI}/src/main.cpp")
add_executable(leaf ${UI_SOURCES} "${LEAF_ROOT}/main/keyboard_config.cpp" "${LEAF_ROOT}/main/keyboard_input.cpp" platform_browser.cpp hardware_services.cpp backlight_browser.cpp)
target_include_directories(leaf PRIVATE compat "${LEAF_ROOT}/main" "${UI}/src" "${UI}/platforms")
target_link_libraries(leaf PRIVATE lvgl)
target_compile_options(leaf PRIVATE -Wno-deprecated-enum-enum-conversion)
target_link_options(leaf PRIVATE -sASYNCIFY -sASYNCIFY_STACK_SIZE=131072 -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=33554432 -sSTACK_SIZE=1048576 -sFORCE_FILESYSTEM=1 -lidbfs.js "-sEXPORTED_FUNCTIONS=['_main','_leaf_report','_leaf_text','_leaf_save','_leaf_state','_leaf_cursor','_leaf_view']" "-sEXPORTED_RUNTIME_METHODS=['ccall','HEAPU8','FS','addRunDependency','removeRunDependency']" -sENVIRONMENT=web -sASSERTIONS=1)
set_target_properties(leaf PROPERTIES SUFFIX ".js" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/site")
foreach(asset index.html browser.js style.css)
  configure_file("${CMAKE_CURRENT_LIST_DIR}/${asset}" "${CMAKE_BINARY_DIR}/site/${asset}" COPYONLY)
endforeach()
